cmake_minimum_required(VERSION 3.16)
project(l2s3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(game src/game.cpp)
target_include_directories(game PUBLIC src include)
add_library(game_benchmarks benchmarks/benchmark.cpp)
target_include_directories(game_benchmarks PUBLIC benchmarks)
target_link_libraries(game_benchmarks PUBLIC game)
add_executable(game_tests tests/game_tests.cpp)
target_link_libraries(game_tests PRIVATE game)
add_executable(l2s3 src/main.cpp src/cli.cpp src/browser.cpp)
target_link_libraries(l2s3 PRIVATE game_benchmarks)
if(MSVC)
    foreach(target game game_benchmarks game_tests l2s3)
        target_compile_options(${target} PRIVATE /utf-8)
    endforeach()
endif()
if(WIN32)
    target_link_libraries(l2s3 PRIVATE shell32)
endif()

enable_testing()
add_test(NAME game_tests COMMAND game_tests)
